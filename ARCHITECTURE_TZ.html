  MENU_GROUPS {
    int id PK
    string slug UK
    text title
    int sort_order
    bool active
  }
  MENU_CATEGORIES {
    int id PK
    string slug UK
    int group_id FK
    text title
    string line "hookah|bar|kitchen"
    int sort_order
    bool active
  }
  PRODUCTS {
    int id PK
    string slug UK
    int category_id FK
    text name
    int price
    text description
    text description_short
    text composition
    numeric portion_value
    string portion_unit
    string prep_time
    text image
    bool visible
    bool available
    int sort_order
  }
  PRODUCT_PAIRINGS {
    int id PK
    int product_id FK "источник"
    int paired_product_id FK "цель"
    string kind "gastro|upsell|alt"
    numeric weight "вес ребра"
    text note
    int sort_order
  }
  TABLES {
    string id PK
    text caption
    int capacity
    bool active
  }
  ORDERS {
    string id PK
    string table_id FK
    text table_caption
    string status "open|closed"
    timestamptz created_at
    timestamptz closed_at
  }
  ORDER_ITEMS {
    int id PK
    string order_id FK
    string product_id "slug-снимок"
    text name
    int qty
    int price
    string line
    string category_id
  }
  VIP_EVENTS {
    string id PK
    string slug UK
    text organization
    date event_date
    int bar_free_limit
    string bar_line
    bool active
  }
  VIP_GUESTS {
    string id PK
    string event_id FK
    string token UK
    text first_name
    text last_name
    int free_used
  }
  VIP_CONSUMPTIONS {
    int id PK
    string guest_id FK
    string product_id
    string line
    bool paid_by_guest
  }
  TOURNAMENTS {
    int id PK
    text title
    int max_slots
    text format
    text fee
    bool registration_open
  }
  TOURNAMENT_APPLICATIONS {
    string id PK
    int tournament_id FK
    string status
    text team_name
    text captain_name
    jsonb players
    jsonb sources
  }
  TICKETS {
    string id PK
    text title
    text description
    string category
    string priority
    string status
    text created_by
  }
  ADMIN_USERS {
    string id PK
    string login UK
    text password_hash
    string role "owner|admin|manager|staff"
    text first_name
    text last_name
  }
  GALLERY_ITEMS {
    string id PK
    text image
    text caption
    int sort_order
  }
  PUSH_SUBSCRIPTIONS {
    int id PK
    text endpoint UK
    text p256dh
    text auth
  }
PK — первичный ключFK — внешний ключUK — уникальный ◆ product_pairings — граф связок (2 FK на products: источник и цель)
